// Add each measure individually. Percentages are fractions; format 0.0%.
Total Loans = COUNTROWS(fact_loan)
Funded Amount = SUM(fact_loan[funded_amnt])
Average Loan Amount = AVERAGE(fact_loan[loan_amnt])
Average Interest Rate = DIVIDE(AVERAGE(fact_loan[int_rate]), 100)
Resolved Loans = SUM(fact_loan[resolved])
Defaulted Loans = SUM(fact_loan[default_flag])
Default Rate = DIVIDE([Defaulted Loans], [Resolved Loans])
Fully Paid Rate = DIVIDE(SUM(fact_loan[fully_paid]), [Resolved Loans])
Current Share = DIVIDE(SUM(fact_loan[current]), [Total Loans])
Interest Received = SUM(fact_loan[total_rec_int])
Payments Received = SUM(fact_loan[total_pymnt])
Average DTI = AVERAGE(fact_loan[dti])
Default Origination Amount = CALCULATE([Funded Amount], fact_loan[default_flag] = 1)
Net Unreturned Principal = SUM(fact_loan[unreturned_principal])
Recovery Rate = DIVIDE(CALCULATE(SUM(fact_loan[recoveries]), fact_loan[default_flag] = 1), SUM(fact_loan[pre_recovery_exposure]))
Net Return Proxy = DIVIDE(SUM(fact_loan[net_cash]), CALCULATE([Funded Amount], fact_loan[resolved] = 1))
Average FICO = BLANK()
